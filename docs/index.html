Quick Start

npm install @sugusdaddy/solana-tx-simulator

// Usage
import { TransactionSimulator } from '@sugusdaddy/solana-tx-simulator';

const simulator = new TransactionSimulator(connection);
const result = await simulator.simulate(transaction, wallet);

console.log(result.summary);        // "Swap 1.5 SOL for 150 USDC via Jupiter"
console.log(result.riskLevel);      // "LOW"
console.log(result.balanceChanges); // [{ token: 'SOL', change: -1.5 }, ...]
